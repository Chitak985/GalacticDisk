// Obsolete values:
// bulgeRadius = sets bulgeScaleX, bulgeScaleY, and bulgeScaleZ
// color1 = sets colorBulge and colorArms
// color2 = sets colorInterarm

GalacticDiskDefinition                    // The config node for spawning a disk, must be top level and can be patched by MM just fine
{
    planet = Eeloo                        // The celestial body whose disk needs to be modified. If it does not exist, things go very bad
    starCount = 250000                    // Number of stars in the galaxy. Do not set this as the actual star number, your computer will die
    galaxyRadius = 200                    // The radius of the galaxy. Used to spread out the stars, if you want to scale it up use objectScale
    diskThickness = 2                     // How thick the disk is vertically. 2 is usually good for any kind of disk (irregular galaxies will need something else)
    minStarSize = 0.07                    // What is the smallest a star can be
    maxStarSize = 0.15                    // What is the biggest a star can be
    armCount = 4                          // How many arms the galaxy has
    armTightness = 4                      // How tight the arms are (high numbers make them be much closer together and might dissolve them)
    armWidth = 0.25                       // How wide the arms are
    interarmFraction = 0.4                // What fraction of the stars is inbetween the arms, 0.05 - 0.95
    armConcentration = 0.35               // How concentrated the arms are, 0.05 - 0.95
    bulgeFraction = 0.2                   // What fraction of the stars is in the bulge, 0.05 - 0.95
    bulgeScaleX = 8                       // X size of the bulge
    bulgeScaleY = 8                       // Y size of the bulge
    bulgeScaleZ = 8                       // Z size of the bulge
    colorBulge = 1.0, 0.92, 0.65          // Color of the bulge
    colorArms = 1.0, 0.92, 0.65           // Color of the arms
    colorInterarm = 0.45, 0.65, 1.0       // Color of the stars between the galactic arms
    armHighlightColor = 0.95, 0.55, 0.55  // The color for randomly highlighting some stars in an arm
    armHighlightChance = 0.06             // Each star that is part of an arm has this chance (6% in this case) to be highlighted with a different color
    objectScale = 1e5                     // The size of the disk in-game. This defaults to 1 but that is very low and the disk would likely end up inside the body
    renderScale = 1e5                     // The render scale. This also defaults to 1 and if your disk is small enough you can omit this,
                                          //however doing so with big disks will make them disappear prematurely. 1e5 seems to be enough
                                          //for most things and setting it higher may make the game lag. The way this works is that
                                          //objectScale is the visible disk scale and renderScale*objectScale is the scale KSP sees to
                                          //understand when to cull the disk. By making the renderScale higher, the disk will seem bigger
                                          //to KSP and it will avoid culling it for much longer
    alignRotationToOrbit = false          // Aligns the disk to be parallel to its orbit. This will ignore rotationX and Z
    rotationY = 45                        // Changes the spin phase of the disk
    rotationX = 0                         // Tilts the disk
    rotationZ = 0                         // Orients which direction the tilt axis points
    rotationPeriod = 1000                 // Seconds to complete one full rotation, defaults to 0 which will make the disk static.
    usesTexture = false                   // Will make the particles the disk uses use a custom texture
    texturePath = YourMod/Textures/e.png  // The texture the disk particles use. Ignored if usesTexture is false
}
